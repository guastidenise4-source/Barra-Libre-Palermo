<script>
  // ============================================================
  // PEGÁ ACÁ LA URL DE TU APPS SCRIPT (Paso 3 del tutorial)
  const API_URL = 'https://script.google.com/macros/s/AKfycbyqawYio7MOmRfLjGhYj-1SfNLcv8QmfPnZjSO5f0fCx0dxKKbLgOqjBEjWZG0DTNZBUQ/exec';
  // ============================================================

  let config = { tanda: 'Tanda 1', precio: 30000, alias: 'tanda1' };

  function formatPrice(n) {
    return '$' + Number(n).toLocaleString('es-AR');
  }
  function applyConfig() {
    document.getElementById('tandaTag').textContent = config.tanda;
    document.getElementById('tandaPill').textContent = config.tanda;
    document.getElementById('priceDisplay').textContent = formatPrice(config.precio);
    document.getElementById('aliasDisplay').textContent = config.alias;
  }
  function saveConfig() {
    config.tanda = document.getElementById('cfgTanda').value.trim() || config.tanda;
    config.precio = document.getElementById('cfgPrecio').value || config.precio;
    config.alias = document.getElementById('cfgAlias').value.trim() || config.alias;
    applyConfig();
    document.getElementById('configPanel').classList.remove('open');
  }
  function toggleConfig() {
    document.getElementById('configPanel').classList.toggle('open');
  }
  function copyAlias() {
    const alias = document.getElementById('aliasDisplay').textContent;
    navigator.clipboard.writeText(alias).catch(() => {});
    event.target.textContent = '¡copiado!';
    setTimeout(() => event.target.textContent = 'copiar', 2000);
  }

  async function submitForm() {
    const nombre = document.getElementById('nombre').value.trim();
    const dni = document.getElementById('dni').value.trim();
    const email = document.getElementById('email').value.trim();
    const emailError = document.getElementById('emailError');
    emailError.style.display = 'none';

    if (!nombre || !dni || !email) {
      alert('Por favor completá todos los campos.');
      return;
    }
    if (!email.includes('@') || !email.includes('.')) {
      emailError.style.display = 'block';
      return;
    }

    const btn = document.getElementById('submitBtn');
    btn.disabled = true;
    btn.textContent = 'Enviando...';

    try {
      await fetch(API_URL, {
        method: 'POST',
        body: JSON.stringify({
          action: 'register',
          nombre, dni, email,
          tanda: config.tanda,
          precio: config.precio
        })
      });
    } catch (e) {
      // Continúa igual aunque falle la red — el invitado ya tiene instrucciones
    }

    document.getElementById('successAlias').textContent = config.alias;
    document.getElementById('formSection').style.display = 'none';
    document.getElementById('successSection').style.display = 'block';
  }

  applyConfig();
</script>
